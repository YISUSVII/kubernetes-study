# Cert-Manager ClusterIssuer - Let's Encrypt Production
# cert-manager v1.21.x: ingressClassName replaces deprecated class field
# External Account Binding (EAB) supported for enterprise ACME providers
apiVersion: cert-manager.io/v1
kind: ClusterIssuer
metadata:
  name: letsencrypt-prod-cluster
spec:
  acme:
    server: https://acme-v02.api.letsencrypt.org/directory
    email: admin@example.com
    privateKeySecretRef:
      name: letsencrypt-prod-key-cluster
    solvers:
    # HTTP-01 challenge via nginx ingress (recommended for most cases)
    - http01:
        ingress:
          ingressClassName: nginx
    # HTTP-01 challenge via Gateway API (recommended for Gateway API users)
    - http01:
        gatewayHTTPRoute:
          parentRefs:
          - name: mygateway
            namespace: default
            kind: Gateway
    # DNS-01 challenge via Cloudflare (wildcard certs)
    - dns01:
        cloudflare:
          apiTokenSecretRef:
            name: cloudflare-api-token-secret
            key: api-token
      selector:
        dnsZones:
        - example.com
