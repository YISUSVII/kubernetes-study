# Gateway API BackendTLSPolicy - GA (v1) since Gateway API v1.6
# Configures TLS from the Gateway proxy to backend Services (east-west TLS)
# This is DIFFERENT from Gateway TLS (which is north-south / client-facing)
# BackendTLSPolicy was promoted to Standard channel in Gateway API v1.3 (2025);
# the API version itself graduated from v1alpha3 to v1 in v1.6 (mid-2026).
# NOTE: v1alpha3 is now deprecated and no longer served by the CRD - use v1.
#
# Use case: Ensure traffic between the Gateway and your backend pods is encrypted
# even inside the cluster (zero-trust networking)
---
# Example 1: Basic BackendTLSPolicy using a cluster CA
apiVersion: gateway.networking.k8s.io/v1
kind: BackendTLSPolicy
metadata:
  name: myapp-backend-tls
  namespace: default
spec:
  # Target the Service that the Gateway routes traffic to
  targetRefs:
  - group: ""
    kind: Service
    name: myapp-service
    sectionName: https   # optional: only apply to this port name
  validation:
    # Use a CA from a ConfigMap to verify the backend's certificate
    caCertificateRefs:
    - kind: ConfigMap
      group: ""
      name: backend-ca-cert
    # The hostname the backend certificate must match
    hostname: myapp-backend.default.svc.cluster.local
    # wellKnownCACertificates can be used for public CAs
    # wellKnownCACertificates: System

---
# Example 2: BackendTLSPolicy for a gRPC service using cert-manager issued cert
# Pair this with a cert-manager Certificate on the backend pod
apiVersion: gateway.networking.k8s.io/v1
kind: BackendTLSPolicy
metadata:
  name: grpc-backend-tls
  namespace: default
spec:
  targetRefs:
  - group: ""
    kind: Service
    name: grpc-backend
  validation:
    caCertificateRefs:
    - kind: ConfigMap
      group: ""
      name: cluster-ca-bundle    # ConfigMap populated by cert-manager trust-manager
    hostname: grpc-backend.default.svc.cluster.local

---
# Supporting ConfigMap: CA certificate bundle
# In production, use cert-manager trust-manager to auto-populate this
apiVersion: v1
kind: ConfigMap
metadata:
  name: backend-ca-cert
  namespace: default
data:
  ca.crt: |
    -----BEGIN CERTIFICATE-----
    # Your CA certificate goes here
    # Generate with: kubectl get secret <tls-secret> -o jsonpath='{.data.ca\.crt}' | base64 -d
    -----END CERTIFICATE-----
