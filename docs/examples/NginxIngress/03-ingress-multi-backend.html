# Nginx Ingress - Multiple Backends with Path-based Routing
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: multi-backend-ingress
spec:
  ingressClassName: nginx
  rules:
  - host: example.com
    http:
      paths:
      - path: /api
        pathType: Prefix
        backend:
          service:
            name: api-service
            port:
              number: 8080
      - path: /admin
        pathType: Prefix
        backend:
          service:
            name: admin-service
            port:
              number: 3000
      - path: /
        pathType: Prefix
        backend:
          service:
            name: web-service
            port:
              number: 80
