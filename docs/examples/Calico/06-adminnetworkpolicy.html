# Kubernetes AdminNetworkPolicy (ANP) - K8s 1.31+ Beta (policy.networking.k8s.io/v1alpha1)
# AdminNetworkPolicy gives cluster admins a way to define network policies that
# CANNOT be overridden by namespace/tenant NetworkPolicies.
# Works alongside standard NetworkPolicy (complements, doesn't replace).
# Supported by: Cilium 1.16+, Calico 3.29+, OVN-Kubernetes 24.06+
#
# Key differences vs NetworkPolicy:
# - Cluster-scoped (not namespaced)
# - Priority-ordered (lower number = higher priority)
# - Explicit Allow/Deny/Pass actions (NetworkPolicy only allows "allow")
# - "Pass" delegates decision to Namespace-scoped NetworkPolicies
---
# Example 1: Platform-level deny all egress to metadata endpoints
# Prevents workloads from accessing cloud metadata APIs (security best practice)
apiVersion: policy.networking.k8s.io/v1alpha1
kind: AdminNetworkPolicy
metadata:
  name: deny-cloud-metadata
spec:
  priority: 10   # lower = evaluated first (0-1000)
  subject:
    namespaces:
      matchExpressions:
      - key: kubernetes.io/metadata.name
        operator: NotIn
        values: [kube-system, kube-public]
  egress:
  - name: deny-aws-metadata
    action: Deny
    to:
    - networks:
      - cidr: 169.254.169.254/32   # AWS/GCP/Azure IMDS endpoint
  - name: deny-azure-metadata
    action: Deny
    to:
    - networks:
      - cidr: 168.63.129.16/32    # Azure metadata endpoint

---
# Example 2: Allow monitoring namespace to scrape all pods
apiVersion: policy.networking.k8s.io/v1alpha1
kind: AdminNetworkPolicy
metadata:
  name: allow-monitoring-scrape
spec:
  priority: 20
  subject:
    # Apply to ALL namespaces (pods being scraped)
    namespaces: {}
  ingress:
  - name: allow-prometheus-scrape
    action: Allow
    from:
    - namespaces:
        matchLabels:
          kubernetes.io/metadata.name: monitoring
    ports:
    - portNumber:
        port: 9090
        protocol: TCP
    - portNumber:
        port: 8080
        protocol: TCP

---
# Example 3: Baseline Namespace Isolation ANP
# Pass (delegate) intra-namespace traffic to tenant NetworkPolicies
apiVersion: policy.networking.k8s.io/v1alpha1
kind: AdminNetworkPolicy
metadata:
  name: baseline-isolation
spec:
  priority: 50
  subject:
    namespaces:
      matchExpressions:
      - key: team
        operator: Exists   # Only apply to namespaces with a 'team' label
  ingress:
  - name: pass-same-namespace
    action: Pass    # Let tenant NetworkPolicies decide same-namespace traffic
    from:
    - sameLabels:
      - key: kubernetes.io/metadata.name   # same namespace
  - name: deny-other-namespaces
    action: Deny
    from:
    - namespaces: {}

---
# Example 4: BaselineAdminNetworkPolicy - lowest priority fallback
# Applied AFTER all ANPs and NetworkPolicies have been evaluated
apiVersion: policy.networking.k8s.io/v1alpha1
kind: BaselineAdminNetworkPolicy
metadata:
  name: default-baseline
spec:
  subject:
    namespaces: {}
  ingress:
  - name: deny-all-cross-namespace-ingress
    action: Deny
    from:
    - namespaces: {}
  egress:
  - name: allow-dns
    action: Allow
    to:
    - namespaces:
        matchLabels:
          kubernetes.io/metadata.name: kube-system
    ports:
    - portNumber:
        port: 53
        protocol: UDP
    - portNumber:
        port: 53
        protocol: TCP
  - name: deny-all-other-egress
    action: Deny
    to:
    - networks:
      - cidr: 0.0.0.0/0
