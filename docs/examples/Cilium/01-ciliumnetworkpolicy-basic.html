# Cilium CiliumNetworkPolicy (CNP) - L3/L4 Policies
# Cilium is the most widely deployed CNI in 2025/2026 (used by AWS EKS, GKE Autopilot, etc.)
# CiliumNetworkPolicy extends standard NetworkPolicy with:
# - L7 (HTTP, gRPC, Kafka, DNS) filtering
# - Identity-based security using security identities (not just IPs)
# - FQDN-based egress policies
# - Port range support
#
# Install Cilium: helm install cilium cilium/cilium -n kube-system
# Requires: Cilium 1.16+ for K8s 1.34/1.35 compatibility
---
# Example 1: Deny all ingress by default (mirrors NetworkPolicy pattern)
apiVersion: "cilium.io/v2"
kind: CiliumNetworkPolicy
metadata:
  name: default-deny-ingress
  namespace: default
spec:
  description: "Default deny all ingress for pods in namespace"
  endpointSelector: {}   # matches ALL pods in namespace
  # No ingress rules = deny all ingress

---
# Example 2: Allow only frontend → backend on specific port (L4)
apiVersion: "cilium.io/v2"
kind: CiliumNetworkPolicy
metadata:
  name: backend-allow-from-frontend
  namespace: default
spec:
  description: "Allow frontend pods to reach backend on port 5000"
  endpointSelector:
    matchLabels:
      app: backend
  ingress:
  - fromEndpoints:
    - matchLabels:
        app: frontend
    toPorts:
    - ports:
      - port: "5000"
        protocol: TCP

---
# Example 3: L7 HTTP policy - allow only specific HTTP methods and paths
apiVersion: "cilium.io/v2"
kind: CiliumNetworkPolicy
metadata:
  name: api-l7-policy
  namespace: default
spec:
  description: "Allow only GET /api/v1 and POST /api/v1/data"
  endpointSelector:
    matchLabels:
      app: api-server
  ingress:
  - fromEndpoints:
    - matchLabels:
        app: frontend
    toPorts:
    - ports:
      - port: "8080"
        protocol: TCP
      rules:
        http:
        - method: "GET"
          path: "/api/v1/.*"
        - method: "POST"
          path: "/api/v1/data"
          headers:
          - "Content-Type: application/json"

---
# Example 4: FQDN-based egress policy (allow specific external domains only)
apiVersion: "cilium.io/v2"
kind: CiliumNetworkPolicy
metadata:
  name: allow-external-egress
  namespace: default
spec:
  description: "Allow pods to call specific external APIs only"
  endpointSelector:
    matchLabels:
      app: myapp
  egress:
  # Allow DNS lookups
  - toEndpoints:
    - matchLabels:
        "k8s:io.kubernetes.pod.namespace": kube-system
        "k8s:k8s-app": kube-dns
    toPorts:
    - ports:
      - port: "53"
        protocol: UDP
      - port: "53"
        protocol: TCP
  # Allow specific external FQDNs
  - toFQDNs:
    - matchName: "api.stripe.com"
    - matchName: "api.sendgrid.com"
    - matchPattern: "*.amazonaws.com"
    toPorts:
    - ports:
      - port: "443"
        protocol: TCP
