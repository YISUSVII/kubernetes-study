# Cilium Hubble - Network Observability
# Hubble is Cilium's built-in observability platform
# Provides flow-level visibility into all network traffic in the cluster
# No code changes needed - eBPF captures everything transparently
---
# HubbleUI Ingress - Expose Hubble UI via an Ingress controller (for local access)
# NOTE: this example uses ingressClassName: nginx / nginx.ingress.kubernetes.io
# annotations for kubernetes/ingress-nginx, which was retired in March 2026
# (see ../../DEPRECATIONS.md). Prefer exposing Hubble UI via a Gateway API
# HTTPRoute + a maintained Gateway API implementation (e.g. Cilium's own Gateway
# API support) instead of a new ingress-nginx deployment.
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: hubble-ui
  namespace: kube-system
  annotations:
    nginx.ingress.kubernetes.io/backend-protocol: "HTTP"
spec:
  ingressClassName: nginx
  rules:
  - host: hubble.internal.example.com
    http:
      paths:
      - path: /
        pathType: Prefix
        backend:
          service:
            name: hubble-ui
            port:
              number: 80

---
# Hubble metrics for Prometheus (ServiceMonitor)
# Requires: Hubble installed with hubble.metrics.enabled=true
apiVersion: monitoring.coreos.com/v1
kind: ServiceMonitor
metadata:
  name: hubble-metrics
  namespace: monitoring
  labels:
    release: kube-prometheus-stack
spec:
  namespaceSelector:
    matchNames:
    - kube-system
  selector:
    matchLabels:
      app.kubernetes.io/name: hubble
  endpoints:
  - port: hubble-metrics
    interval: 30s
    path: /metrics
    relabelings:
    - sourceLabels: [__meta_kubernetes_pod_node_name]
      targetLabel: node

---
# Example Flow Query (run with hubble CLI):
# Monitor HTTP traffic in real time:
#   hubble observe --namespace default --protocol http
#
# Watch all DNS queries:
#   hubble observe --namespace default --protocol dns
#
# Check connectivity between pods:
#   hubble observe --from-pod default/frontend --to-pod default/backend
#
# Export flows to file:
#   hubble observe -o json > flows.json
