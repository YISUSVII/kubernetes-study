# KEDA ScaledObject - CPU-based Scaling
# Requires: metrics-server installed in cluster
# KEDA v2.15+ compatible - uses explicit apiVersion/kind in scaleTargetRef
apiVersion: keda.sh/v1alpha1
kind: ScaledObject
metadata:
  name: cpu-scaler
  namespace: default
spec:
  scaleTargetRef:
    apiVersion: apps/v1
    kind: Deployment
    name: myapp
  pollingInterval: 15  # seconds between trigger checks
  cooldownPeriod: 60   # seconds before scaling down after last active trigger
  idleReplicaCount: 0  # scale to 0 when idle (KEDA v2.12+)
  minReplicaCount: 1
  maxReplicaCount: 10
  fallback:             # what to do if trigger source is unavailable
    failureThreshold: 3
    replicas: 3
  triggers:
  - type: cpu
    metricType: Utilization
    metadata:
      value: "60"
